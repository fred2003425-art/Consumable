    function initDatabase() {
      document.getElementById('authOverlay').classList.add('hidden');

      const categories = [...new Set(items.map(i => i.category))].filter(Boolean);

    function validateSuffixInput() {
      const input = document.getElementById('itemSuffix');
      input.value = input.value.replace(/\D/g, '');
      const isValid = input.value.length === 8;
      document.getElementById('codeError').classList.toggle('hidden', isValid || !input.value);

        const matchesCat = !catFilter || item.category === catFilter;
        return matchesSearch && matchesCat;
      });

      if (filtered.length === 0) {
        container.innerHTML = `<div class="text-center py-16 bg-white rounded-2xl border border-slate-100"><p class="text-slate-500 text-sm">目前沒有符合條件的零食貨品</p></div>`;

    function openModal(editMode = false) {
      if (!editMode) {
        document.getElementById('itemForm').reset();
        document.getElementById('editingItemId').value = '';
        document.getElementById('modalTitle').innerText = '新增零食貨品';
        clearImagePreview();
      }
      document.getElementById('itemModal').classList.remove('hidden');
    }

    function closeModal() {
      document.getElementById('itemModal').classList.add('hidden');
    }

    function editItem(id) {
      const item = items.find(i => i.id === id);
      if (!item) return;

      document.getElementById('editingItemId').value = id;
      document.getElementById('modalTitle').innerText = '編輯零食貨品';
      document.getElementById('itemSuffix').value = item.code ? item.code.replace('4000', '') : '';
      document.getElementById('itemName').value = item.name || '';
      document.getElementById('itemCategory').value = item.category || '';
      document.getElementById('itemPieceCount').value = item.pieceCount || '';
      
      if (item.image && item.image.startsWith('data:image')) {
        currentBase64Image = item.image;
        document.getElementById('itemImage').value = '';
        setPreviewImageSrc(item.image);
      } else {
        currentBase64Image = '';
        document.getElementById('itemImage').value = item.image || '';
        setPreviewImageSrc(item.image || '');
      }

      openModal(true);
    }

    // 處理本地上傳圖片
    function handleFileUpload(e) {
      const file = e.target.files[0];
      if (file) {
        processImageFile(file);
      }
    }

    // 處理 Ctrl+V 貼上剪貼簿截圖
    function handlePasteImage(e) {
      if (document.getElementById('itemModal').classList.contains('hidden')) return;
      const items = (e.clipboardData || e.originalEvent.clipboardData).items;
      for (let item of items) {
        if (item.kind === 'file' && item.type.startsWith('image/')) {
          const blob = item.getAsFile();
          processImageFile(blob);
          break;
        }
      }
    }

    // 將圖片轉為 Base64（並進行適當壓縮）
    function processImageFile(file) {
      const reader = new FileReader();
      reader.onload = function(event) {
        const img = new Image();
        img.onload = function() {
          const canvas = document.createElement('canvas');
          let width = img.width;
          let height = img.height;
          const maxDim = 800; // 限制最大邊長以節省雲端空間

          if (width > maxDim || height > maxDim) {
            if (width > height) {
              height = Math.round((height * maxDim) / width);
              width = maxDim;
            } else {
              width = Math.round((width * maxDim) / height);
              height = maxDim;
            }
          }

          canvas.width = width;
          canvas.height = height;
          const ctx = canvas.getContext('2d');
          ctx.drawImage(img, 0, 0, width, height);

          currentBase64Image = canvas.toDataURL('image/jpeg', 0.8);
          document.getElementById('itemImage').value = '';
          setPreviewImageSrc(currentBase64Image);
        };
        img.src = event.target.result;
      };
      reader.readAsDataURL(file);
    }

    function previewImageSrc(url) {
      currentBase64Image = '';
      setPreviewImageSrc(url);
    }

    function setPreviewImageSrc(src) {
      const container = document.getElementById('imagePreviewContainer');
      const img = document.getElementById('imagePreview');
      if (src) {
        img.src = src;
        container.classList.remove('hidden');
      } else {
        container.classList.add('hidden');
      }
    }

    function clearImagePreview() {
      currentBase64Image = '';
      document.getElementById('itemImage').value = '';
      document.getElementById('imageFileInput').value = '';
      document.getElementById('imagePreviewContainer').classList.add('hidden');
    }

    function saveItem(e) {
      e.preventDefault();
      if (!validateSuffixInput()) return;

      const editId = document.getElementById('editingItemId').value;
      const imageUrl = currentBase64Image || document.getElementById('itemImage').value;

      const itemData = {
        code: '4000' + document.getElementById('itemSuffix').value,
        name: document.getElementById('itemName').value,
        category: document.getElementById('itemCategory').value,
        pieceCount: document.getElementById('itemPieceCount').value,
        image: imageUrl || ''
      };

      if (editId) {
        // 更新現有商品
        db.child(editId).update(itemData).then(() => closeModal());
      } else {
        // 新增商品
        db.push(itemData).then(() => closeModal());
      }
    }

    function deleteItem(id) {
      if (confirm('確定要刪除這筆零食資料嗎？')) {
        db.child(id).remove();
      }
    }
